configuration
3 Imports & hyperparameters
import os, json, torch import numpy as np from torch import nn from torch.utils.data import Dataset, DataLoader from transformers import CLIPModel, CLIPProcessor, get_cosine_schedule_with_warmup from PIL import Image import torchvision.transforms as T PROJECT_DIR = "/content/mercadona_clip" DATASET_PATH = f"{PROJECT_DIR}/mercadona_dataset.json" IMAGES_DIR = f"{PROJECT_DIR}/images" MODEL_DIR = f"{PROJECT_DIR}/model" os.makedirs(MODEL_DIR, exist_ok=True) CFG = { "model_name" : "openai/clip-vit-base-patch32", "batch_size" : 64, "epochs" : 20, "lr" : 5e-6, # low LR — fine-tuning pretrained weights "weight_decay": 0.01, "warmup_steps": 100, "grad_clip" : 1.0, "val_split" : 0.1, "patience" : 5, "freeze" : "last2", # projections + last 2 transformer blocks per encoder } DEVICE = "cuda" if torch.cuda.is_available() else "cpu" print(f"Device: {DEVICE}") print(f"GPU: {torch.cuda.get_device_name(0)}")
dataset
4 Dataset class, train/val split & dataloaders
class MercadonaDataset(Dataset): def __init__(self, items, processor, augment=False): self.items = items self.processor = processor self.aug = T.Compose([ T.RandomHorizontalFlip(), T.ColorJitter(0.2, 0.2, 0.1), T.RandomAffine(degrees=5, translate=(0.05, 0.05)), ]) if augment else None def __len__(self): return len(self.items) def __getitem__(self, idx): item = self.items[idx] img = Image.open(item["image_path"]).convert("RGB") if self.aug: img = self.aug(img) inputs = self.processor( text=[item["text"]], images=img, return_tensors="pt", padding="max_length", truncation=True, max_length=77 ) return {k: v.squeeze(0) for k, v in inputs.items()} with open(DATASET_PATH) as f: all_items = json.load(f) all_items = [p for p in all_items if p["image_path"] and os.path.exists(p["image_path"])] print(f"Valid items: {len(all_items)}") np.random.seed(42) np.random.shuffle(all_items) split = int(len(all_items) * (1 - CFG["val_split"])) train_items, val_items = all_items[:split], all_items[split:] print(f"Train: {len(train_items)} | Val: {len(val_items)}") processor = CLIPProcessor.from_pretrained(CFG["model_name"]) train_ds = MercadonaDataset(train_items, processor, augment=True) val_ds = MercadonaDataset(val_items, processor, augment=False) train_dl = DataLoader(train_ds, batch_size=CFG["batch_size"], shuffle=True, num_workers=2, pin_memory=True) val_dl = DataLoader(val_ds, batch_size=CFG["batch_size"], shuffle=False, num_workers=2, pin_memory=True)
model
5 Load model & apply freeze strategy
model = CLIPModel.from_pretrained(CFG["model_name"]).to(DEVICE) def freeze_model(model, strategy): # Freeze all parameters first for p in model.parameters(): p.requires_grad = False if strategy == "heads": # Only projection layers — 0.4% trainable (run 1, Recall@1=56.1%) for p in model.visual_projection.parameters(): p.requires_grad = True for p in model.text_projection.parameters(): p.requires_grad = True elif strategy == "last2": # Projections + last 2 transformer blocks — ~3-4% trainable (run 2, Recall@1=66.7%) for p in model.visual_projection.parameters(): p.requires_grad = True for p in model.text_projection.parameters(): p.requires_grad = True for layer in model.vision_model.encoder.layers[-2:]: for p in layer.parameters(): p.requires_grad = True for layer in model.text_model.encoder.layers[-2:]: for p in layer.parameters(): p.requires_grad = True trainable = sum(p.numel() for p in model.parameters() if p.requires_grad) total = sum(p.numel() for p in model.parameters()) print(f"Strategy : {strategy}") print(f"Trainable: {trainable:,} / {total:,} ({100*trainable/total:.1f}%)") freeze_model(model, CFG["freeze"])
training setup
6 Loss function & optimizer
def clip_loss(img_emb, txt_emb, logit_scale): img_emb = nn.functional.normalize(img_emb, dim=-1) txt_emb = nn.functional.normalize(txt_emb, dim=-1) scale = logit_scale.exp().clamp(max=100) logits = (img_emb @ txt_emb.T) * scale labels = torch.arange(len(logits), device=DEVICE) loss_i = nn.functional.cross_entropy(logits, labels) loss_t = nn.functional.cross_entropy(logits.T, labels) return (loss_i + loss_t) / 2 optimizer = torch.optim.AdamW( filter(lambda p: p.requires_grad, model.parameters()), lr=CFG["lr"], weight_decay=CFG["weight_decay"], betas=(0.9, 0.98), eps=1e-6 ) total_steps = len(train_dl) * CFG["epochs"] scheduler = get_cosine_schedule_with_warmup( optimizer, CFG["warmup_steps"], total_steps ) print(f"Total steps: {total_steps} | Warmup: {CFG['warmup_steps']}")
training
7 Training loop with early stopping ~45–60 min on T4
best_val, patience_count, history = float("inf"), 0, [] for epoch in range(CFG["epochs"]): # --- train --- model.train() train_loss = 0 for batch in train_dl: batch = {k: v.to(DEVICE) for k, v in batch.items()} out = model(**batch) loss = clip_loss(out.image_embeds, out.text_embeds, model.logit_scale) loss.backward() torch.nn.utils.clip_grad_norm_(model.parameters(), CFG["grad_clip"]) optimizer.step(); scheduler.step(); optimizer.zero_grad() train_loss += loss.item() train_loss /= len(train_dl) # --- validate --- model.eval() val_loss = 0 with torch.no_grad(): for batch in val_dl: batch = {k: v.to(DEVICE) for k, v in batch.items()} out = model(**batch) val_loss += clip_loss(out.image_embeds, out.text_embeds, model.logit_scale).item() val_loss /= len(val_dl) history.append({"epoch": epoch+1, "train": train_loss, "val": val_loss}) print(f"Epoch {epoch+1:02d} | train {train_loss:.4f} | val {val_loss:.4f}") if val_loss < best_val: best_val, patience_count = val_loss, 0 model.save_pretrained(MODEL_DIR) processor.save_pretrained(MODEL_DIR) print(f" ✓ Saved best model (val={best_val:.4f})") else: patience_count += 1 print(f" patience {patience_count}/{CFG['patience']}") if patience_count >= CFG["patience"]: print("Early stopping"); break print(f"\n✅ Best val loss: {best_val:.4f}")

9 Recall@K evaluation
model_eval = CLIPModel.from_pretrained(MODEL_DIR).to(DEVICE).eval() processor = CLIPProcessor.from_pretrained(MODEL_DIR) img_embs, txt_embs = [], [] with torch.no_grad(): for batch in val_dl: batch = {k: v.to(DEVICE) for k, v in batch.items()} out = model_eval(**batch) img_embs.append(nn.functional.normalize(out.image_embeds, dim=-1).cpu()) txt_embs.append(nn.functional.normalize(out.text_embeds, dim=-1).cpu()) img_embs = torch.cat(img_embs) txt_embs = torch.cat(txt_embs) sims = txt_embs @ img_embs.T def recall_at_k(sims, k): topk = sims.topk(k, dim=1).indices correct = (topk == torch.arange(len(sims)).unsqueeze(1)).any(dim=1) return correct.float().mean().item() print(f"Recall@1 : {recall_at_k(sims,1)*100:.1f}%") print(f"Recall@5 : {recall_at_k(sims,5)*100:.1f}%") print(f"Recall@10: {recall_at_k(sims,10)*100:.1f}%")

10 Query test
from deep_translator import GoogleTranslator def translate(text): try: return GoogleTranslator(source="es", target="en").translate(text) except: return text for query in ["leche entera", "aceite de oliva", "yogur natural"]: translated = translate(query) inputs = processor(text=[translated], return_tensors="pt", padding=True).to(DEVICE) with torch.no_grad(): q_emb = model_eval.get_text_features(**inputs) if not isinstance(q_emb, torch.Tensor): q_emb = q_emb.pooler_output q_emb = nn.functional.normalize(q_emb, dim=-1).cpu() scores = (q_emb @ img_embs.T).squeeze() top3 = scores.topk(3).indices.tolist() print(f"\n'{query}' → '{translated}'") for idx in top3: item = val_items[idx] print(f" [{scores[idx]:.3f}] {item['name']} — {item['category']}")
Results achieved: Recall@1 66.7% · Recall@5 93.3% · Recall@10 96.1% · Model: openai/clip-vit-base-patch32 · Freeze: last2 · Dataset: 4,620 Mercadona products
Known limitation: The model is English-only. At inference, Spanish queries are translated to English via deep-translator before being passed to CLIP. Translation is inconsistent — some queries work better untranslated. Next step: retrain with translated Spanish product names in the dataset, or find a working multilingual CLIP model.
